/etc/nginx/conf.d/learn.conf
server {
    listen 80;
    server_name nginx-lab.local;

    location / {
        root /var/www/html;
        index index.html;
        try_files $uri $uri/ =404;
    }

    location /status {
        stub_status on;
        access_log off;
    }

    location ~ \.css$ {
        add_header Content-Type text/css;
    }
}

location /secret { return 418 "Я чайник! 🫖"; add_header X-NGINX-Master true; }

docker exec nginx-ubuntu nginx -s reload
